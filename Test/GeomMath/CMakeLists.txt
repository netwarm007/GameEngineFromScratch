cmake_minimum_required(VERSION 4.0)
project(GeomMathRegression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(USE_ISPC OFF)
configure_file("${root}/config.h.in" "${CMAKE_CURRENT_BINARY_DIR}/config.h")

add_executable(GeomMathCppTest "${root}/Test/Framework/GeomMathTest.cpp")
target_include_directories(GeomMathCppTest PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
    "${root}/Framework/GeomMath" "${root}/Framework/Common" "${root}/Framework/Algorism")
add_test(NAME GeomMathCppTest COMMAND GeomMathCppTest)

if(WIN32)
    set(CMAKE_ISPC_COMPILER "${root}/External/Windows/bin/ispc.exe")
else()
    set(CMAKE_ISPC_COMPILER "${root}/External/${CMAKE_SYSTEM_NAME}/bin/ispc")
endif()
enable_language(ISPC)
if(NOT CMAKE_ISPC_INSTRUCTION_SETS)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64|ARM64")
        set(CMAKE_ISPC_INSTRUCTION_SETS neon-i32x4)
    else()
        set(CMAKE_ISPC_INSTRUCTION_SETS sse2-i32x4 avx2-i32x8)
    endif()
endif()
add_subdirectory("${root}/Framework/GeomMath/ispc" "${CMAKE_CURRENT_BINARY_DIR}/ispc")
add_executable(GeomMathIspcTest "${root}/Test/Framework/GeomMathTest.cpp")
target_include_directories(GeomMathIspcTest PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
    "${root}/Framework/GeomMath" "${root}/Framework/Common" "${root}/Framework/Algorism")
target_compile_definitions(GeomMathIspcTest PRIVATE USE_ISPC)
target_link_libraries(GeomMathIspcTest PRIVATE GeomMath)
add_test(NAME GeomMathIspcTest COMMAND GeomMathIspcTest)
